??? quote "Stub code in `ideIntegration.sdsstub`"

    ```sds linenums="93"
    @Experimental
    enum DSPipelineActivity {
    // Pre-Processing Layer
        /**
         * Acquire data: load from external sources, or build a datatype by hand.
         */
        DataAcquisitionQDataLoading
        DataAcquisitionQDatatypeConstruction
        DataAcquisitionQUtilities

        /**
         * Prepare data before the split: deterministic cleaning, schema edits, EDA, helpers.
         */
        DataPreparationQExploration
        DataPreparationQPreSplitCleaning
        DataPreparationQSchemaModification
        DataPreparationQUtilities

        /**
         * Partition data into training, validation and test sets.
         */
        DataPartitioningQDataSplitting

        /**
         * Process data after the split. Exploration, post-split cleaning and augmentation are training-only.
         */
        DataProcessingQExploration
        DataProcessingQPostSplitCleaning
        DataProcessingQSchemaModification
        DataProcessingQUtilities
        DataProcessingQDataTransformation
        DataProcessingQAugmentation

    // Model Building Layer
        /**
         * Do feature engineering to prepare your data for modeling.
         */
        FeatureEngineeringQDatatypeConstruction
        FeatureEngineeringQSchemaModification
        FeatureEngineeringQUtilities
        FeatureEngineeringQEngineering
        FeatureEngineeringQFeatureTransformation

        /**
         * Select features and assemble the dataset for your model.
         */
        FeatureSelectionQSchemaModification
        FeatureSelectionQTabularDatasetConversion

        /**
         * Configure a model.
         */
        ModelingQModelCreation

        /**
         * Train the model on the training set.
         */
        TrainingQModelFitting

        /**
         * Predict and score on the validation set, then once on the test set to assess generality.
         */
        EvaluationQPrediction
        EvaluationQMetricCalculation

        TestingQPrediction
        TestingQMetricCalculation

        /**
         * Interpret results: inverse-transform predictions back to original units, visualize the model.
         */
        InterpretationQPostProcessing
        InterpretationQVisualization

        /**
         * Can be used as a wildcard.
         */
        Any
    }
    ```
